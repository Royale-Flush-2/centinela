## sql:
```sql
SELECT c.nombre AS cedi,
       i.cajas_disponibles,
       v.venta_diaria_prom_14d,
       ROUND(i.cajas_disponibles / v.venta_diaria_prom_14d, 1) AS cobertura_dias
FROM inventario_cedi i
JOIN cedis c ON c.id = i.cedi_id
JOIN v_venta_diaria v ON v.sku = i.sku AND v.cedi_id = i.cedi_id
WHERE i.sku = '70412'
  AND i.fecha_corte = '2026-10-02 06:00'
ORDER BY cobertura_dias;
```
